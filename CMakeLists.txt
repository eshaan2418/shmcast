cmake_minimum_required(VERSION 3.25)
project(shmcast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SHMCAST_SANITIZE "" CACHE STRING "address,undefined | thread | empty")
option(SHMCAST_FUZZ "Build the libFuzzer reader target (clang with fuzzer runtime)" OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(SHMCAST_SANITIZE)
  add_compile_options(-fsanitize=${SHMCAST_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SHMCAST_SANITIZE})
endif()
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
add_library(shmcast INTERFACE)
target_include_directories(shmcast INTERFACE include)
target_link_libraries(shmcast INTERFACE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(shmcast INTERFACE rt)
endif()
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE SYSTEM)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
foreach(t gtest gtest_main gmock gmock_main)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE -w)
  endif()
endforeach()
enable_testing()
file(GLOB TEST_SOURCES tests/test_*.cpp)
add_executable(shmcast-tests ${TEST_SOURCES})
target_link_libraries(shmcast-tests PRIVATE shmcast GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(shmcast-tests DISCOVERY_TIMEOUT 120 PROPERTIES TIMEOUT 300)
add_executable(shmcast-bench tools/bench.cpp)
target_link_libraries(shmcast-bench PRIVATE shmcast)
add_executable(shmcast-itch tools/itch.cpp)
target_link_libraries(shmcast-itch PRIVATE shmcast ZLIB::ZLIB)
if(SHMCAST_FUZZ)
  add_executable(fuzz-reader fuzz/fuzz_reader.cpp)
  target_compile_options(fuzz-reader PRIVATE -fsanitize=fuzzer,address,undefined -g)
  target_link_options(fuzz-reader PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(fuzz-reader PRIVATE shmcast)
endif()
